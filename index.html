import hashlib
import hmac

def generate_mines(server_seed, client_seed, nonce, mine_count):
    message = f"{client_seed}-{nonce}".encode('utf-8')
    key = server_seed.encode('utf-8')
    
    hash_result = hmac.new(key, message, hashlib.sha256).hexdigest()
    
    positions = list(range(25))
    grid = ['diamond'] * 25
    
    hash_index = 0
    for i in range(mine_count):
        if hash_index + 8 > len(hash_result):
            new_msg = f"{client_seed}-{nonce}-{i}".encode('utf-8')
            hash_result = hmac.new(key, new_msg, hashlib.sha256).hexdigest()
            hash_index = 0
            
        sub_hash = hash_result[hash_index:hash_index+8]
        hash_index += 8
        
        random_num = int(sub_hash, 16)
        chosen_index = random_num % len(positions)
        
        actual_pos = positions[chosen_index]
        grid[actual_pos] = 'mine'
        positions.pop(chosen_index)
        
    return grid

print(generate_mines("YOUR_SERVER_SEED", "YOUR_CLIENT_SEED", 1, 3))
